  function updateCharCount(input) {
    const counter = document.getElementById('char-counter');
    if (counter) {
      counter.innerText = input.value.length + '/120';
    }
  }

  function selectCondition(el, type) {
    const chips = document.querySelectorAll('.condition-chip');
    chips.forEach(chip => {
      chip.className = 'condition-chip p-3 rounded-xl bg-surface-container-lowest text-left shadow-sm active:scale-[0.98] transition-all';
      const title = chip.querySelector('.font-title');
      if (title) title.className = 'font-title text-[14px] font-semibold text-on-surface';
      const desc = chip.querySelector('p');
      if (desc) desc.className = 'font-body-sm text-[11px] text-secondary mt-1';
      const indicator = chip.querySelector('.chip-indicator');
      if (indicator) {
        indicator.className = 'w-4 h-4 rounded-full bg-surface-container-high chip-indicator';
        indicator.innerText = '';
      }
    });

    el.className = 'condition-chip active-chip p-3 rounded-xl bg-inverse-surface text-inverse-on-surface text-left shadow-sm active:scale-[0.98] transition-all';
    const activeTitle = el.querySelector('.font-title');
    if (activeTitle) activeTitle.className = 'font-title text-[14px] font-semibold text-inverse-on-surface';
    const activeDesc = el.querySelector('p');
    if (activeDesc) activeDesc.className = 'font-body-sm text-[11px] text-inverse-on-surface/80 mt-1';
    const activeIndicator = el.querySelector('.chip-indicator');
    if (activeIndicator) {
      activeIndicator.className = 'w-4 h-4 rounded-full bg-primary flex items-center justify-center text-[11px] text-on-primary chip-indicator';
      activeIndicator.innerText = '✓';
    }
    showToast("Đã cập nhật tình trạng: " + (activeTitle ? activeTitle.innerText : ''));
  }

  function applyAiSuggestions() {
    const titleInput = document.getElementById('product-title');
    if (titleInput) {
      titleInput.value = "Kệ inox 4 tầng đựng gia vị & đĩa nhà hàng inox 201";
      updateCharCount(titleInput);
    }
    
    const priceInput = document.getElementById('input-selling-price');
    if (priceInput) {
      priceInput.value = "1.450.000";
    }

    const appliedFeedback = document.getElementById('ai-applied-feedback');
    if (appliedFeedback) {
      appliedFeedback.classList.remove('hidden');
    }

    showToast("✨ AI đã tối ưu hóa tiêu đề, phân loại & giá bán!");
  }

  function dismissAiBadge() {
    showToast("Bạn có thể điều chỉnh thủ công từng trường thông tin.");
  }

  function formatCurrency(input) {
    let val = input.value.replace(/\D/g, "");
    if (val) {
      input.value = new Intl.NumberFormat('vi-VN').format(val);
    }
  }

  function saveDraftNotification() {
    showToast("Đã lưu bản nháp vào danh sách quản lý kho.");
  }

  function proceedToStep3() {
    showToast("Chuyển sang Bước 3: Kiểm duyệt & Xác nhận bài đăng...");
  }

  function showToast(msg) {
    const toast = document.getElementById('toast');
    const toastMsg = document.getElementById('toast-msg');
    if (!toast || !toastMsg) return;
    
    toastMsg.innerText = msg;
    toast.classList.remove('opacity-0', 'translate-y-3', 'pointer-events-none');
    toast.classList.add('opacity-100', 'translate-y-0');

    setTimeout(() => {
      toast.classList.remove('opacity-100', 'translate-y-0');
      toast.classList.add('opacity-0', 'translate-y-3', 'pointer-events-none');
    }, 2400);
  }
